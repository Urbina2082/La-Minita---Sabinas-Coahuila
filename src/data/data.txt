<section class="menu">

  <!-- CATEGORÍAS -->
  <div class="categorias">

    {menu.map((categoria, index) => (
      <button
        class:list={{ activo: index === 0 }}
        class="categoria-btn"
        data-categoria={categoria.categoria}
      >
        {categoria.categoria}
      </button>
    ))}

  </div>


  <!-- PRODUCTOS -->
  <div class="productos" id="productos">

    {menu[0].items.map((item) => (

      <article class="producto">

        <h3>{item.nombre}</h3>

        {item.descripcion && (
          <p>{item.descripcion}</p>
        )}

        {item.precio && (
          <strong>
            ${item.precio}.00
          </strong>
        )}

      </article>

    ))}

  </div>

</section>


<script define:vars={{ menu }}>

  const botones = document.querySelectorAll(".categoria-btn");
  const productos = document.querySelector("#productos");


  function mostrarCategoria(nombreCategoria) {

    const categoria = menu.find(
      item => item.categoria === nombreCategoria
    );

    if (!categoria) return;


    productos.innerHTML = categoria.items.map(item => {

      let precio = "";

      if (item.precio) {

        precio = `
          <strong>
            $${item.precio}.00
          </strong>
        `;

      }

      return `
        <article class="producto">

          <h3>${item.nombre}</h3>

          ${
            item.descripcion
              ? `<p>${item.descripcion}</p>`
              : ""
          }

          ${precio}

        </article>
      `;

    }).join("");

  }


  botones.forEach(boton => {

    boton.addEventListener("click", () => {

      botones.forEach(btn =>
        btn.classList.remove("activo")
      );

      boton.classList.add("activo");

      mostrarCategoria(
        boton.dataset.categoria
      );

    });

  });

</script>
